\section{Course 1: Divide and Conquer}


\begin{definition}

A technique that solves a problem by:

\begin{itemize}

\item dividing the problem into smaller sub-problems;

\item recursively solving each part;

\item combining the obtained solutions.

\end{itemize}

\end{definition}


\separator % The separation  bar


\textbf{General principle}

A recursive algorithm usually contains:

\begin{itemize}

\item a base case;

\item a reduction step;

\item a combination step.

\end{itemize}



\begin{formula}

Recurrence relation:

\[
T(n)=aT(\frac{n}{b})+f(n)
\]

where:

\begin{itemize}

\item $a$: number of sub-problems;

\item $b$: reduction factor;

\item $f(n)$: combination cost.

\end{itemize}

\end{formula}



\begin{example}

Merge Sort:

\[
T(n)=2T(n/2)+O(n)
\]


The array is divided into two halves.

The merge operation combines sorted sub-arrays.


Final complexity:

\[
O(n\log n)
\]


\end{example}

\section{Course 2: Complexity Analysis}


\begin{important}

Algorithm complexity describes how the execution time grows
according to the input size.

\end{important}



\textbf{Common complexity classes}

\begin{itemize}

\item $O(1)$ : constant

\item $O(\log n)$ : logarithmic

\item $O(n)$ : linear

\item $O(n\log n)$ : quasi-linear

\item $O(n^2)$ : quadratic

\end{itemize}

\section{Course 3: Classical Algorithms}


\begin{example}

Binary Search:


Sorted input:

\[
[1,3,5,8,10,15]
\]


The search space is divided by two
at each iteration.


Complexity:

\[
O(\log n)
\]


\end{example}



\begin{warning}

Binary search requires
a sorted input array.

\end{warning}
