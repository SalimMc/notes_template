\textbf{Core concepts}


\begin{itemize}

\item \textit{Linear independence: vectors cannot be expressed as combinations of others.}

\item \textit{Basis: smallest set of vectors generating the whole space.}

\item \textit{Dimension: number of vectors in a basis.}

\item \textit{Span: all possible linear combinations.}

\end{itemize}



\vspace{3mm}



\textbf{Important formula}


\[
A^{-1}
=
\frac{1}{\det(A)}
\operatorname{adj}(A)
\]



\vspace{2mm}



\textbf{Key condition}


\textit{
A matrix is invertible if and only if
$\det(A)\neq0$.
}
